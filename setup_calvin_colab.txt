# ============================================================
# CALVIN Setup for Google Colab
# Prerequisites: MemoryVLA already installed (Python 3.10, memvla env)
# This installs CALVIN into the existing memvla environment
# ============================================================

# Step 1: Clone CALVIN repository
%%bash
cd /content
git clone --recurse-submodules https://github.com/mees/calvin.git
export CALVIN_ROOT=/content/calvin
echo "✅ CALVIN repository cloned"

# Step 2: Install CALVIN in memvla environment (Python 3.10)
%%bash
source /opt/conda/etc/profile.d/conda.sh
conda activate memvla

cd /content/calvin

# Install wheel and cmake
pip install wheel
pip install cmake==3.18.4

# Install tacto (tactile sensing, can skip for inference-only)
cd calvin_env/tacto
pip install -e .

# Install calvin_env
cd /content/calvin/calvin_env
pip install -e .

# Install calvin_models (for policy loading if needed)
cd /content/calvin/calvin_models
pip install -e .

# Downgrade setuptools if pyhash installation fails
pip install 'setuptools<58'

echo "✅ CALVIN installed in memvla environment (Python 3.10)"
