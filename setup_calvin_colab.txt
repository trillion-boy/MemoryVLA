# ============================================================
# CALVIN Setup for Google Colab
# ============================================================

# Step 1: Install Miniforge (Python 3.8 environment)
%%bash
wget https://github.com/conda-forge/miniforge/releases/latest/download/Miniforge3-Linux-x86_64.sh
bash Miniforge3-Linux-x86_64.sh -b -p /opt/conda
rm Miniforge3-Linux-x86_64.sh

# Step 2: Create Calvin environment
%%bash
source /opt/conda/etc/profile.d/conda.sh
conda create -n calvin_venv python=3.8 -y
conda activate calvin_venv

# Verify Python version
python --version

# Step 3: Clone CALVIN repository
%%bash
cd /content
git clone --recurse-submodules https://github.com/mees/calvin.git
export CALVIN_ROOT=/content/calvin

# Step 4: Install dependencies
%%bash
source /opt/conda/etc/profile.d/conda.sh
conda activate calvin_venv

cd /content/calvin

# Install wheel and cmake
pip install wheel
pip install cmake==3.18.4

# Install tacto (if needed, can skip for inference)
cd calvin_env/tacto
pip install -e .

# Install calvin_env
cd /content/calvin/calvin_env
pip install -e .

# Install calvin_models (for policy loading)
cd /content/calvin/calvin_models
pip install -e .

# Downgrade setuptools if pyhash fails
pip install 'setuptools<58'

# Step 5: Download dataset (debug version for testing)
%%bash
source /opt/conda/etc/profile.d/conda.sh
conda activate calvin_venv

cd /content/calvin/dataset
sh download_data.sh debug

# This downloads ~1.3GB debug dataset
# For full dataset: sh download_data.sh D (or ABC, ABCD)

# Step 6: Test installation
%%bash
source /opt/conda/etc/profile.d/conda.sh
conda activate calvin_venv

cd /content/calvin

python -c "
import gym
import calvin_env
print('✅ CALVIN environment installed successfully!')

# Test environment creation
env = gym.make('calvin_env/playdata/task_D_D/validation')
print(f'✅ Environment created: {env}')
"
